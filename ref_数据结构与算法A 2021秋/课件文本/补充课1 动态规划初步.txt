===== 幻灯片 24 =====
24
0-1背包问题–“记忆化”递归实现
int dp_knapsack(int w[], int v[], int c, int f[][100], int n)
// f[n][c]表示背包可选物品为1,2,..i，容量为c时的最优解
{
if (f[n][c] != -1) return f[n][c]; // 保证只被计算一次
if (n == 0) return f[n][c] = 0;
f[n][c] = dp_knapsack(w, v, c, f, n - 1);
if (c - w[n - 1] >= 0)
f[n][c] = max(f[n][c], v[n - 1] + dp_knapsack(w, v, c - w[n - 1], f, n - 1));
return f[n][c];
}

===== 幻灯片 30 =====

0-1背包问题– 递推实现
int dp_knapsack(int w[], int v[], int W, int f[][100], int n)
{
for (int j = 0; j <= W; j++)
f[0][j] = 0;
for (int i = 1; i <= n; i++)
for (int j = 0; j <= W; ++j) {
if (j – w[i-1] >= 0)
f[i][j] = max(f[i-1][j], f[i-1][j-w[i-1]] + v[i-1]);
else
f[i][j] = f[i-1][j];
}
return f[n][W];
}

===== 幻灯片 34 =====
34
0-1背包问题– 递推实现空间优化
int dp_knapsack(int w[], int v[], int W, int f[], int n)
{
for (int j = 0; j <= W; j++)
f[j] = 0;
for (int i = 1; i <= n; i++)
for (int j = W; j >= 0; ++j) { // 从大到小枚举，保证f[i-1][j-w[i-1]]还未被改写
if (j – w[i-1] >= 0)
f[j] = max(f[[j], f[j-w[i-1]] + v[i-1]);
}
return f[W];
}

字符串编辑距离
[edit distance] 字符串的编辑距离，又称为Levenshtein距离，由俄罗斯的数学家Vladimir Levenshtein在1965年提出。是指利用字符操作，把字符串A转换成字符串B所需要的最少操作数。其中，字符操作包括：
删除一个字符 e.g. abcd  abd
插入一个字符 e.g. abd  abcd
修改一个字符 e.g. abcd  abed
问：给定两个字符串A和B，长度分别为n和m，求字符串A至少经过多少步字符操作变成字符串B。

int edit_distance(char A[], char B[], int f[][100], int n, int m)
{
if (f[n][m] != -1) return f[n][m]; // 保证只被计算一次
if (n == 0) return f[n][m] = m;
else if (m == 0) return f[n][m] = n;
else if (A[i-1] == B[j-1])
return f[n][m] = edit_distance(A, B, f, n-1, m-1);
else
return f[n][m] = min(edit_distance(A, B, f, n, m-1) + 1,
min(edit_distance(A, B, n-1, m) + 1,
edit_distance(A, B, n-1, m-1) + 1));
}

===== 幻灯片 42 =====

字符串编辑距离–递推实现
int edit_distance(char A[], char B[], int f[][100], int n, int m)
{
for (int i = 0; i <= n; i++) f[i][0] = i;
for (int j = 0; j <= m; j++) f[0][j] = j;
for (int i = 1; i <= n; i++)
for (int j = 1; j <= m; j++) {
if (A[i-1] == B[j-1])
f[i][j] = f[i-1][j-1];
else
f[i][j] = min(f[i][j-1]+1, min(f[i-1][j]+1, f[i-1][j-1]+1));
}
return f[n][m];
}
